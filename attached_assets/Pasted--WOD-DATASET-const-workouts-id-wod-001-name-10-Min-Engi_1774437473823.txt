// ===== SIMPLE PARSER =====
function parseInput(input) {
  const text = input.toLowerCase();

  const movements = [
    "running", "rowing", "ski erg", "bike",
    "wall balls", "burpees", "lunges",
    "dumbbell", "sled", "sit-ups", "plank"
  ];

  let include = [];
  let exclude = [];

  movements.forEach(m => {
    if (text.includes(m)) include.push(m);
  });

  if (text.includes("no")) {
    movements.forEach(m => {
      if (text.includes("no " + m)) exclude.push(m);
    });
  }

  let duration = null;
  const match = text.match(/\d+/);
  if (match) duration = parseInt(match[0]);

  return { include, exclude, duration };
}


// ===== MATCHING ENGINE =====
function scoreWorkout(wod, filters) {
  let score = 0;

  // include scoring
  filters.include.forEach(i => {
    if (wod.exercises.some(e => e.includes(i))) score += 5;
  });

  // exclude filter
  for (let ex of filters.exclude) {
    if (wod.exercises.some(e => e.includes(ex))) return -1;
  }

  // duration scoring
  if (filters.duration) {
    const diff = Math.abs(wod.duration - filters.duration);
    if (diff <= 2) score += 3;
    else if (diff <= 5) score += 1;
  }

  return score;
}


// ===== MAIN FUNCTION =====
function getWODs(input) {
  const filters = parseInput(input);

  const results = workouts
    .map(w => ({ ...w, score: scoreWorkout(w, filters) }))
    .filter(w => w.score >= 0)
    .sort((a, b) => b.score - a.score)
    .slice(0, 3);

  return results;
}
