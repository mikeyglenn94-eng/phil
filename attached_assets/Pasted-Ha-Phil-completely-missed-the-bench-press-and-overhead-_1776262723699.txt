async function scanSessionsForInjury(clientId, bodyPart, windowDays) {
  const sessions = await getSessionsInWindow(clientId, windowDays);
  const affected = [];
  
  const relevantKeywords = injuryMap[bodyPart] || [];
  
  for (const session of sessions) {
    for (const exercise of session.exercises) {
      const nameLC = exercise.name.toLowerCase();
      const isAffected = relevantKeywords.some(keyword => 
        nameLC.includes(keyword.toLowerCase())
      );
      if (isAffected) {
        affected.push({
          sessionDate: session.date,
          sessionName: session.name,
          exerciseName: exercise.name
        });
      }
    }
  }
  
  return affected;
}
